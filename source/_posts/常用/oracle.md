COMMENT ON COLUMN UT_OFFERING_EXAM_FORM.CREATE_BY IS '创建人';
COMMENT ON COLUMN UT_OFFERING_EXAM_FORM.CREATE_TIME IS '创建时间';
COMMENT ON COLUMN UT_OFFERING_EXAM_FORM.UPDATE_BY IS '更新人';
COMMENT ON COLUMN UT_OFFERING_EXAM_FORM.UPDATE_TIME IS '更新时间';

# 物化视图

```
CREATE MATERIALIZED VIEW "CQDX_JXGLXX"."PROGRAM_COURSE_ALL_V" 
  ORGANIZATION HEAP PCTFREE 10 PCTUSED 40 INITRANS 1 MAXTRANS 255 
 NOCOMPRESS LOGGING
  STORAGE(INITIAL 65536 NEXT 1048576 MINEXTENTS 1 MAXEXTENTS 2147483645
  PCTINCREASE 0 FREELISTS 1 FREELIST GROUPS 1
  BUFFER_POOL DEFAULT FLASH_CACHE DEFAULT CELL_FLASH_CACHE DEFAULT)
  TABLESPACE "YG_JXGLXX" 
  BUILD IMMEDIATE
  USING INDEX 
  REFRESH COMPLETE ON DEMAND START WITH sysdate+0 NEXT SYSDATE + (30/(24*60))
  USING DEFAULT LOCAL ROLLBACK SEGMENT
  USING ENFORCED CONSTRAINTS DISABLE QUERY REWRITE
  AS 
(
-- 课程
SELECT
 PCV.GRADE, PCV.MAJOR_ID, pcv.program_type,  PCV.COURSE_CODE, PCV.MODULE_ID REAL_MODULE_ID,
 PCV.COURSE_VER_IND_ID,
 PCV.MODULE_NAME REAL_MODULE_NAME,
 
 CASE PCV.REQUIRED_FLAG
WHEN '必修' THEN 'Y'
ELSE 'N' END REAL_REQUIRED_FLAG,
 '1培养方案课程' AS CATE
FROM PROGRAM_COURSE_V PCV
UNION
-- 课程集
SELECT
 PCV.GRADE, PCV.MAJOR_ID, pcv.program_type,  PCV.COURSE_CODE, PCV.MODULE_ID REAL_MODULE_ID,
  PCV.MODULE_NAME REAL_MODULE_NAME,
	PCV.COURSE_VER_IND_ID,
 CASE PCV.REQUIRED_FLAG
WHEN '必修' THEN 'Y'
ELSE 'N' END REAL_REQUIRED_FLAG,
 '2培养方案课程集' AS CATE
FROM PROGRAM_COURSE_SET_V PCV
union
-- 课程替换
SELECT
 PCV.GRADE, PCV.MAJOR_ID, pcv.program_type,  UC2.CODE COURSE_CODE, PCV.MODULE_ID REAL_MODULE_ID,
  PCV.COURSE_VER_IND_ID,
	PCV.MODULE_NAME REAL_MODULE_NAME,
 CASE PCV.REQUIRED_FLAG
WHEN '必修' THEN 'Y'
ELSE 'N' END REAL_REQUIRED_FLAG,
 '3培养方案课程替换' AS CATE
FROM PROGRAM_COURSE_V PCV
	JOIN ENRL_COURSE_REPLACE_SETTING  ECRS
ON ECRS.ORIGIN_VER_INT_ID = PCV.COURSE_VER_IND_ID
JOIN UNITIME_WORKFLOW_STATUS UWS
ON UWS.UNIQUEID = ECRS.REVIEW_STATUS_ID
JOIN UNITIME_COURSE UC2
ON UC2.VER_IND_ID = ECRS.REPLACE_VER_INT_ID
AND UC2.COURSE_STATUS IN ('已启用', '停开')
and uws.status_flag = '通过'
UNION
-- 课程集替换
SELECT
 PCV.GRADE, PCV.MAJOR_ID, pcv.program_type,  UC2.CODE COURSE_CODE, PCV.MODULE_ID REAL_MODULE_ID,
 	PCV.COURSE_VER_IND_ID,
  PCV.MODULE_NAME REAL_MODULE_NAME,

 CASE PCV.REQUIRED_FLAG
WHEN '必修' THEN 'Y'
ELSE 'N' END REAL_REQUIRED_FLAG,
 '4培养方案课程集替换' AS CATE
FROM PROGRAM_COURSE_SET_V PCV
	JOIN ENRL_COURSE_REPLACE_SETTING  ECRS
ON ECRS.ORIGIN_VER_INT_ID = PCV.COURSE_VER_IND_ID
JOIN UNITIME_WORKFLOW_STATUS UWS
ON UWS.UNIQUEID = ECRS.REVIEW_STATUS_ID
JOIN UNITIME_COURSE UC2
ON UC2.VER_IND_ID = ECRS.REPLACE_VER_INT_ID
AND UC2.COURSE_STATUS IN ('已启用', '停开')
and uws.status_flag = '通过'
UNION
-- 通识教育课程
SELECT
 CP.GRADE, TO_NUMBER(CP.MAJOR) MAJOR_ID, CPC.NAME program_type,  CCSD.COURSE_CODE COURSE_CODE, CCMD.UNIQUEID REAL_MODULE_ID,
  UC.VER_IND_ID COURSE_VER_IND_ID,
  CCMD.NAME REAL_MODULE_NAME,
 'N' AS  REAL_REQUIRED_FLAG,
 '5通识课程集' AS CATE
FROM CM_PROGRAM CP
JOIN CM_COURSE_SET CCS
ON CCS.COURSE_CATEGORY = '通识教育课程'
JOIN CM_COURSE_MODULE_CATEGORY CCMD
ON CCMD.NAME ='通识教育课程'
JOIN CM_COURSE_SET_DETAIL CCSD
ON CCS.UNIQUEID = CCSD.COURSE_SET_ID
JOIN UNITIME_COURSE UC 
ON UC.CODE = CCSD.COURSE_CODE
AND UC.COURSE_STATUS IN ('已启用', '停开')
JOIN CM_PROGRAM_CATEGORY CPC
ON CPC.UNIQUEID = CP.PROGRAM_CATE_ID
WHERE CP.PROGRAM_STATUS = '激活'
UNION
-- 非限制先修课程课程
SELECT
 CP.GRADE, TO_NUMBER(CP.MAJOR) MAJOR_ID, CPC.NAME program_type,  CCSD.COURSE_CODE COURSE_CODE, CCMD.UNIQUEID REAL_MODULE_ID,
 UC.VER_IND_ID COURSE_VER_IND_ID,
  CCMD.NAME REAL_MODULE_NAME,
 'N' AS  REAL_REQUIRED_FLAG,
 '6非限课程集' AS CATE
FROM CM_PROGRAM CP
JOIN CM_COURSE_SET CCS
ON CCS.COURSE_CATEGORY = '非限制选修课程'
JOIN CM_COURSE_MODULE_CATEGORY CCMD
ON CCMD.NAME ='个性化模块'
JOIN CM_COURSE_SET_DETAIL CCSD
ON CCS.UNIQUEID = CCSD.COURSE_SET_ID
JOIN UNITIME_COURSE UC 
ON UC.CODE = CCSD.COURSE_CODE
AND UC.COURSE_STATUS IN ('已启用', '停开')
JOIN CM_PROGRAM_CATEGORY CPC
ON CPC.UNIQUEID = CP.PROGRAM_CATE_ID
WHERE CP.PROGRAM_STATUS = '激活'
)
```
